def _make_grid_skeleton(spacing=950.0, clip_r=2800.0):
    pts = []
    s = spacing
    for i in range(-8, 9):
        for j in range(-8, 9):
            p = np.array([s * (i + 0.5 * (j % 2)), s * np.sqrt(3) / 2 * j])
            if np.linalg.norm(p) <= clip_r:
                pts.append(p)
    return np.array(pts)

### 3.3 policy.py 改动

#### 改动 1：ring_cover → grid_cover

```python
def grid_cover(self):
    """31点格点证书扫描（替代 Q3 的 ring_cover）.

    策略：
    1. 对每个格点，扫描所有 unknown 频道
    2. 免费扫描 detected 频道
    3. 检出新源 → 立即 hunt_all
    4. 自适应格点数：unknown 少时只需部分格点
    5. 贪心选最近格点
    """
    grid_pts = [SKELETON[i] for i in range(len(SKELETON))]
    visited = set()

    while len(visited) < len(grid_pts):
        unknown = self.ledger.unknown_channels()
        if not unknown:
            break
        if len(self.sim.cleared) >= N_CHANNELS:
            break
        if self.sim.elapsed_real() > REAL_TIME_LIMIT:
            break

        # 贪心选最近未访问格点
        cur = self.ledger.cur_pos
        best_d = float('inf')
        best_j = -1
        for j, pt in enumerate(grid_pts):
            if j in visited:
                continue
            d = np.linalg.norm(pt - cur)
            if d < best_d:
                best_d = d
                best_j = j
        if best_j < 0:
            break
        visited.add(best_j)
        target = grid_pts[best_j]

        if not self._is_visited(target, tol=100):
            self._visited.append(np.asarray(target, float))

        # 扫描所有 unknown 频道（按频道接近度排序省换频）
        unknown_sorted = sorted(unknown,
            key=lambda c: abs(c - self.ledger.cur_channel))
        for ch in unknown_sorted:
            if ch in self.sim.cleared:
                continue
            if not self.ledger.channels[ch].is_active:
                continue
            resp = self.sim.measure(target, ch)
            self._handle_measure(ch, target, resp)
            self._sync()

        # 免费扫描 detected 频道
        self._free_scan_at(target)

        # 检查 clearable
        for ch in list(unknown):
            self.ledger.check_clearable(ch)

        # 顺路清除 + 追踪新发现源
        self._try_enroute(target)
        self.hunt_all()
        self._mark_empty_grid(visited)
```

### 3.5 mock 测试中的模拟器改造

mock 测试需要模拟定向源。需要在 mock 模拟器中增加：
- 每个源随机生成类型（全向/定向）
- 定向源随机生成朝向 φ
- 检测逻辑：距离 ≤ R **且** |ang(G→S) - φ| ≤ 90° 才返回 direction，否则 no_signal

```python
# mock 模拟器中的检测逻辑
def measure(self, pos, ch):
    # ... 移动到 pos, 切换频道 ...
    src = self.sources.get(ch)
    if src is None:
        return {"measure_result": "no_signal"}
    d = np.linalg.norm(src.pos - pos)
    if d <= NEAR_R:
        return {"measure_result": "near"}
    if d > src.R:
        return {"measure_result": "no_signal"}
    # Q4 新增：定向源盲区检查
    if src.is_directional:
        angle_to_sensor = np.degrees(np.arctan2(pos[1] - src.pos[1], pos[0] - src.pos[0]))
        diff = abs(((angle_to_sensor - src.phi + 180) % 360) - 180)
        if diff > 90:
            return {"measure_result": "no_signal"}  # 在盲区内
    # 有信号：返回示向度 + 误差
    true_bearing = np.degrees(np.arctan2(src.pos[1] - pos[1], src.pos[0] - pos[0]))
    bearing = true_bearing + src.error[pos]  # ±1° 误差
    return {"measure_result": "direction", "svd_deg": bearing}
```

### v3：自适应格点缩减

**触发条件**：v2 测试显示格点移动成本占比 > 60%。

**改动**：
- 利用 F_k^omni（全向假设收缩后的可行域）跳过"该区域已排除"的格点
- 利用朝向区间跳过"朝向不可能"的格点
- 格点按 2-opt + 贪心最近邻排序

### v4：类型判别 + 朝向累积

**触发条件**：v3 测试显示定向源追踪效率低或格点缩减不足。

**改动**：
- N7 全向判别引理：追踪途中的多停点方向差 > 180° 都检出 → 全向
- 朝向区间累积：每次检出给 180° 弧约束，多次检出自然缩窄
- 朝向缩窄后进一步缩减格点
